// xs weathering/patches [Kind] [Amount] [Scale] [Seed]
//
// xs_begin
// author : 'Rok'
// arg : { name = 'Kind'    var = 'm_kind'    range = '0 3'     value = '0'    step = '1'  precision = '0' }
// arg : { name = 'Amount'  var = 'm_amount'  range = '0 100'   value = '40'   step = '1'  precision = '0' }
// arg : { name = 'Scale'   var = 'm_scale'   range = '1 32'    value = '4'    step = '1'  precision = '0' }
// arg : { name = 'Seed'    var = 'm_seed'    range = '0 9999'  value = '0'    step = '1'  precision = '0' }
// xs_end
// rok_about : 'Patches of rust, dirt, plaster or verdigris'
// rok_colours : { start = '0'  rgb = '8B3A1A B5602A 5A4632 7A6246 D9D2C3 BDB5A5 4F9A86 76BBA6' }
// Rok puts these colours in the palette from the selected colour on: two for
// rust, dirt, plaster and verdigris each.
// rok_tip Kind : 0 is rust, 1 is dirt, 2 is plaster and 3 is verdigris. Rust and dirt gather at edges and low down. Plaster and verdigris spread evenly.
// rok_tip Amount : How much of the surface gets covered, in percent. 0 still leaves a few small spots. 100 covers about ten times more than 0.
// rok_tip Scale : How big a patch is, in cells. 1 gives fine spots. 32 gives large areas.
// rok_tip Seed : Picks another random layout of the patches.

// Paints patches onto the outside of the model. Kind 0 is rust, 1 dirt, 2
// plaster, 3 verdigris (green copper). Rust and dirt gather at edges and
// low down. Amount is how much of the surface they cover, Scale how big a
// patch is. It only recolours voxels on the surface.

float pa_hash3(vec3 p3) {
  p3 = fract(p3 * 0.1031);
  p3 += dot(p3, p3.zyx + 31.32);
  return fract((p3.x + p3.y) * p3.z);
}

float pa_noise(vec3 p) {
  vec3 i = floor(p);
  vec3 f = p - i;
  vec3 u = f * f * (3.0 - 2.0 * f);
  float a = mix(pa_hash3(i), pa_hash3(i + vec3(1.0, 0.0, 0.0)), u.x);
  float b = mix(pa_hash3(i + vec3(0.0, 1.0, 0.0)), pa_hash3(i + vec3(1.0, 1.0, 0.0)), u.x);
  float c = mix(pa_hash3(i + vec3(0.0, 0.0, 1.0)), pa_hash3(i + vec3(1.0, 0.0, 1.0)), u.x);
  float d = mix(pa_hash3(i + vec3(0.0, 1.0, 1.0)), pa_hash3(i + vec3(1.0, 1.0, 1.0)), u.x);
  return mix(mix(a, b, u.y), mix(c, d, u.y), u.z);
}

float map(vec3 v) {
  float here = voxel(v);
  if (here == 0.0) return 0.0;
  float open = 0.0;
  if (voxel(v + vec3(1.0, 0.0, 0.0)) == 0.0) open += 1.0;
  if (voxel(v - vec3(1.0, 0.0, 0.0)) == 0.0) open += 1.0;
  if (voxel(v + vec3(0.0, 1.0, 0.0)) == 0.0) open += 1.0;
  if (voxel(v - vec3(0.0, 1.0, 0.0)) == 0.0) open += 1.0;
  if (voxel(v + vec3(0.0, 0.0, 1.0)) == 0.0) open += 1.0;
  if (voxel(v - vec3(0.0, 0.0, 1.0)) == 0.0) open += 1.0;
  if (open == 0.0) return here;
  vec3 p = floor(v) + 0.5;
  vec3 seed = vec3(m_seed * 2.7, m_seed * 1.3, m_seed * 3.9);
  float kind = floor(m_kind + 0.5);
  float n = pa_noise(p / m_scale + seed) * 0.65 + pa_noise(p / (m_scale * 0.4) + seed * 2.0) * 0.35;
  float lure = 0.0;
  if (kind < 1.5) {
    // Rust and dirt: more at edges and corners, and low down.
    lure = (open - 1.0) * 0.05 + (1.0 - p.z / i_volume_size.z) * 0.08;
  }
  if (n - lure > 0.25 + m_amount / 100.0 * 0.45) return here;
  return i_color_index + kind * 2.0 + (pa_noise(p / 1.3 + seed) > 0.55 ? 1.0 : 0.0);
}
