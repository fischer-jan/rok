// xs weathering/cracks [Amount] [Size] [Depth] [Seed]
//
// xs_begin
// author : 'Rok'
// arg : { name = 'Amount'  var = 'm_amount'  range = '0 100'   value = '45'   step = '1'  precision = '0' }
// arg : { name = 'Size'    var = 'm_size'    range = '3 24'    value = '7'    step = '1'  precision = '0' }
// arg : { name = 'Depth'   var = 'm_depth'   range = '0 1'     value = '1'    step = '1'  precision = '0' }
// arg : { name = 'Seed'    var = 'm_seed'    range = '0 9999'  value = '0'    step = '1'  precision = '0' }
// xs_end
// rok_about : 'Cracks in stone, plaster or dry ground'
// rok_tip Amount : How much of the surface is cracked, in patches, in percent. 0 gives no cracks. 100 cracks all of it.
// rok_tip Size : The size of one piece between the cracks, in cells. 3 gives many small pieces. 24 gives few large ones.
// rok_tip Depth : 1 cuts the cracks one voxel deep. It cuts only on flat faces with two cells of model behind them, so a plate less than 3 cells thick is only darkened. 0 only darkens the cracks. Edges and corners are only darkened in both cases.
// rok_tip Seed : Picks another random layout of the cracks.

// Cracks the surface of the model, as old stone, plaster, ice or dry
// ground cracks: thin lines that meet in a net of uneven pieces. Size is
// how big a piece is, Amount how much of the surface is cracked (in
// patches; 100 cracks it all). Depth 1 cuts the cracks one voxel deep and
// darkens the bottom of them; Depth 0 only paints them dark.
//
// The net comes from a Voronoi pattern on each face (see ck_crack), with
// only some of its lines cracked. Edges and corners are only darkened, never
// cut, so the outline of the model stays. The dark is a darker shade of
// each voxel's own colour, the nearest the palette has.

float ck_hash3(vec3 p3) {
  p3 = fract(p3 * 0.1031);
  p3 += dot(p3, p3.zyx + 31.32);
  return fract((p3.x + p3.y) * p3.z);
}

float ck_noise(vec3 p) {
  vec3 i = floor(p);
  vec3 f = p - i;
  vec3 u = f * f * (3.0 - 2.0 * f);
  float a = mix(ck_hash3(i), ck_hash3(i + vec3(1.0, 0.0, 0.0)), u.x);
  float b = mix(ck_hash3(i + vec3(0.0, 1.0, 0.0)), ck_hash3(i + vec3(1.0, 1.0, 0.0)), u.x);
  float c = mix(ck_hash3(i + vec3(0.0, 0.0, 1.0)), ck_hash3(i + vec3(1.0, 0.0, 1.0)), u.x);
  float d = mix(ck_hash3(i + vec3(0.0, 1.0, 1.0)), ck_hash3(i + vec3(1.0, 1.0, 1.0)), u.x);
  return mix(mix(a, b, u.y), mix(c, d, u.y), u.z);
}

// True when the voxel at v lies on a crack. The cracks are drawn in the
// plane of the face the voxel shows: (x + y, z) on a wall, so they run on
// around corners, and (x, y) on a floor or a ceiling. In that plane the
// pieces are Voronoi cells: points scattered one per square of Size cells,
// and a crack wherever a cell is about as near to two of them.
bool ck_crack(vec3 v) {
  vec3 c = floor(v);
  // Patches: only where the low noise lets it.
  if (ck_noise((c + 0.5 + m_seed * 7.1) / (m_size * 2.5)) * 100.0 < 100.0 - m_amount * 1.25) return false;
  bool side = voxel(v + vec3(1.0, 0.0, 0.0)) == 0.0 || voxel(v - vec3(1.0, 0.0, 0.0)) == 0.0
    || voxel(v + vec3(0.0, 1.0, 0.0)) == 0.0 || voxel(v - vec3(0.0, 1.0, 0.0)) == 0.0;
  bool capped = voxel(v + vec3(0.0, 0.0, 1.0)) == 0.0 || voxel(v - vec3(0.0, 0.0, 1.0)) == 0.0;
  float face = capped && !side ? 1.0 : 0.0;
  vec2 q = (face > 0.5 ? c.xy : vec2(c.x + c.y, c.z)) + 0.5;
  vec2 box = floor(q / m_size);
  float d1 = 1e9;
  float d2 = 1e9;
  vec2 b1 = box;
  vec2 b2 = box;
  for (int i = -1; i <= 1; i++) {
    for (int j = -1; j <= 1; j++) {
      vec2 b = box + vec2(float(i), float(j));
      vec3 h = vec3(b, face + m_seed * 0.01);
      vec2 point = (b + vec2(ck_hash3(h), ck_hash3(h + 3.7)) * 0.8 + 0.1) * m_size;
      float d = length(point - q);
      if (d < d1) { d2 = d1; b2 = b1; d1 = d; b1 = b; }
      else if (d < d2) { d2 = d; b2 = b; }
    }
  }
  // Only some sides of the pieces crack, so the lines branch and end
  // instead of closing into a net of tiles. The pair of points names a side.
  if (ck_hash3(vec3(b1 + b2, face + m_seed * 0.01) + 0.37) > 0.6) return false;
  // d2 - d1 is twice the distance to the line between the two points.
  return d2 - d1 < 1.1;
}

bool ck_surface(vec3 v) {
  return voxel(v + vec3(1.0, 0.0, 0.0)) == 0.0 || voxel(v - vec3(1.0, 0.0, 0.0)) == 0.0
    || voxel(v + vec3(0.0, 1.0, 0.0)) == 0.0 || voxel(v - vec3(0.0, 1.0, 0.0)) == 0.0
    || voxel(v + vec3(0.0, 0.0, 1.0)) == 0.0 || voxel(v - vec3(0.0, 0.0, 1.0)) == 0.0;
}

// A darker shade of the colour in slot here: the nearest palette colour to
// it times 0.55, among the colours that really are darker. Hue counts twice
// as much as brightness. Here itself when the palette has none.
float ck_darker(float here) {
  vec3 c = palette(here).rgb;
  vec3 w = vec3(0.299, 0.587, 0.114);
  float lc = dot(c, w);
  vec3 want = c * 0.55;
  float best = here;
  float best_d = 1e9;
  for (int i = 1; i < 256; i++) {
    vec3 q = palette(float(i)).rgb;
    if (dot(q, w) > lc - 0.04) continue;
    vec3 e = q - want;
    float l = dot(e, w);
    vec3 hue = e - l;
    float dist = l * l + 2.0 * dot(hue, hue);
    if (dist < best_d) { best_d = dist; best = float(i); }
  }
  return best;
}

float map(vec3 v) {
  float here = voxel(v);
  if (here == 0.0) return 0.0;
  if (ck_surface(v)) {
    if (!ck_crack(v)) return here;
    // Cut it, but only on a flat face (open on one side: edges and corners
    // stay, so the outline of the model does not change) where the model is
    // thick enough behind it (a thin wall keeps its skin: no holes).
    if (m_depth > 0.5) {
      float open = 0.0;
      for (int s = 0; s < 6; s++) {
        vec3 d = s == 0 ? vec3(1.0, 0.0, 0.0) : s == 1 ? vec3(-1.0, 0.0, 0.0) : s == 2 ? vec3(0.0, 1.0, 0.0)
               : s == 3 ? vec3(0.0, -1.0, 0.0) : s == 4 ? vec3(0.0, 0.0, 1.0) : vec3(0.0, 0.0, -1.0);
        if (voxel(v + d) == 0.0) open += 1.0;
      }
      if (open < 1.5) for (int s = 0; s < 6; s++) {
        vec3 d = s == 0 ? vec3(1.0, 0.0, 0.0) : s == 1 ? vec3(-1.0, 0.0, 0.0) : s == 2 ? vec3(0.0, 1.0, 0.0)
               : s == 3 ? vec3(0.0, -1.0, 0.0) : s == 4 ? vec3(0.0, 0.0, 1.0) : vec3(0.0, 0.0, -1.0);
        // Open on this side, and two filled cells behind it.
        if (voxel(v + d) == 0.0 && voxel(v - d) > 0.0 && voxel(v - 2.0 * d) > 0.0) return 0.0;
      }
    }
    return ck_darker(here);
  }
  // Inside: the bottom of a cut crack, when a surface voxel next to it is cut.
  if (m_depth < 0.5) return here;
  for (int s = 0; s < 6; s++) {
    vec3 d = s == 0 ? vec3(1.0, 0.0, 0.0) : s == 1 ? vec3(-1.0, 0.0, 0.0) : s == 2 ? vec3(0.0, 1.0, 0.0)
           : s == 3 ? vec3(0.0, -1.0, 0.0) : s == 4 ? vec3(0.0, 0.0, 1.0) : vec3(0.0, 0.0, -1.0);
    vec3 n = v + d;
    if (voxel(n) > 0.0 && voxel(n + d) == 0.0 && voxel(v - d) > 0.0 && ck_crack(n)) return ck_darker(here);
  }
  return here;
}
