// xs weathering/stains [Length] [Amount] [Width] [Seed]
//
// xs_begin
// author : 'Rok'
// arg : { name = 'Length'  var = 'm_length'  range = '1 64'    value = '10'   step = '1'  precision = '0' }
// arg : { name = 'Amount'  var = 'm_amount'  range = '0 100'   value = '50'   step = '1'  precision = '0' }
// arg : { name = 'Width'   var = 'm_width'   range = '1 8'     value = '1'    step = '1'  precision = '0' }
// arg : { name = 'Seed'    var = 'm_seed'    range = '0 9999'  value = '0'    step = '1'  precision = '0' }
// xs_end
// rok_about : 'Rain streaks running down the walls'
// rok_colours : { start = '0'  rgb = '3E3A36 6A625A' }
// Rok puts these colours in the palette from the selected colour on: a dark
// and a faded grime. They are only used when the palette has no darker shade
// of a wall's own colour.
// rok_tip Length : How far the longest streak runs down the wall, in cells. Shorter streaks run down to about a quarter of Length. Streaks start at the top of a wall face.
// rok_tip Amount : How many streaks there are, in percent of the wall columns. 0 gives none. Streaks appear only on walls that are at least 3 cells tall.
// rok_tip Width : How many cells wide each streak is. 1 gives thin lines. 8 gives broad bands.
// rok_tip Seed : Picks another random layout of the streaks.

// Streaks that run down the walls, as rain leaves them: from the top edge
// of a wall, from under a ledge and from under a window. A streak is dark
// at the top and fades lower down, and its end breaks up. Length is how far
// the longest ones run, Amount how many there are, Width how wide.
//
// A streak is a darker shade of the wall's own colour: the nearest colour
// the palette has. So it works on any wall colour. It only recolours voxels
// of walls, and only of walls at least three voxels tall, so the steps of a
// roof or the rim of a floor plate stay clean.

float st_hash(vec3 p3) {
  p3 = fract(p3 * 0.1031);
  p3 += dot(p3, p3.zyx + 31.32);
  return fract((p3.x + p3.y) * p3.z);
}

// The cells of the wall face above v (up is 1) or below it (up is -1): they
// count while they are filled and open to side d too.
float st_run(vec3 v, vec3 d, float up, float most) {
  float k = 0.0;
  for (int i = 1; i <= 64; i++) {
    if (float(i) > most) break;
    vec3 c = v + vec3(0.0, 0.0, up * float(i));
    if (c.z < 0.0 || c.z >= i_volume_size.z || voxel(c) == 0.0 || voxel(c + d) > 0.0) break;
    k += 1.0;
  }
  return k;
}

// A darker shade of the colour in slot here: the palette colour nearest to
// it scaled by f, among the colours that really are darker. Hue counts
// twice as much as brightness, so the shade keeps its hue when it can.
float st_darker(float here, float f) {
  vec3 c = palette(here).rgb;
  vec3 w = vec3(0.299, 0.587, 0.114);
  float lc = dot(c, w);
  vec3 want = c * f;
  float best = here;
  float best_d = 1e9;
  for (int i = 1; i < 256; i++) {
    vec3 q = palette(float(i)).rgb;
    if (dot(q, w) > lc - 0.04) continue;
    vec3 e = q - want;
    float l = dot(e, w);
    vec3 hue = e - l;
    float dist = l * l + 2.0 * dot(hue, hue);
    if (dist < best_d) { best_d = dist; best = float(i); }
  }
  return best;
}

float map(vec3 v) {
  float here = voxel(v);
  if (here == 0.0) return 0.0;
  for (int s = 0; s < 4; s++) {
    vec3 d = s == 0 ? vec3(1.0, 0.0, 0.0) : (s == 1 ? vec3(-1.0, 0.0, 0.0)
           : (s == 2 ? vec3(0.0, 1.0, 0.0) : vec3(0.0, -1.0, 0.0)));
    if (voxel(v + d) > 0.0) continue;
    // Streaks run in columns along the wall, Width cells wide. The face
    // (its side and its layer) is part of the column's name.
    float along = d.x != 0.0 ? floor(v.y) : floor(v.x);
    float layer = d.x != 0.0 ? floor(v.x) : floor(v.y);
    vec3 id = vec3(floor(along / m_width), float(s) * 131.0 + layer, m_seed);
    if (st_hash(id) * 100.0 >= m_amount) continue;
    float len = m_length * (0.35 + 0.65 * st_hash(id + 5.3));
    // Each column of a wide streak runs a little shorter or longer.
    len *= 0.75 + 0.25 * st_hash(vec3(along, id.y, m_seed) + 9.1);
    float k = st_run(v, d, 1.0, len + 1.0);  // how far below the top of the face
    if (k >= len) continue;
    if (k + 1.0 + st_run(v, d, -1.0, 2.0) < 3.0) continue;  // too short a face
    // The last third breaks up into drops.
    float tail = (k / len - 0.66) / 0.34;
    if (tail > 0.0 && st_hash(floor(v) + m_seed * 0.13) < tail) continue;
    return st_darker(here, k < len * 0.4 ? 0.62 : 0.8);
  }
  return here;
}
