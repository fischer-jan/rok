// xs hard_surface/battle_damage [Size] [Amount] [Scorch] [Glow] [Holes] [Seed]
//
// xs_begin
// author : 'Rok'
// arg : { name = 'Size'    var = 'm_size'    range = '1 8'     value = '3'    step = '1'  precision = '0' }
// arg : { name = 'Amount'  var = 'm_amount'  range = '0 100'   value = '60'   step = '1'  precision = '0' }
// arg : { name = 'Scorch'  var = 'm_scorch'  range = '0 100'   value = '40'   step = '1'  precision = '0' }
// arg : { name = 'Glow'    var = 'm_glow'    range = '0 100'   value = '30'   step = '1'  precision = '0' }
// arg : { name = 'Holes'   var = 'm_holes'   range = '0 1'     value = '0'    step = '1'  precision = '0' }
// arg : { name = 'Seed'    var = 'm_seed'    range = '0 9999'  value = '0'    step = '1'  precision = '0' }
// xs_end
// rok_colours : { start = '0'  rgb = '1C1B1A 37332F 5A534B FF8A2A'  names = 'Soot, Dark scorch, Ash, Glow' }
// rok_about : 'Hits on a hull: dents, soot and glowing rims'
// rok_tip Size : The radius of the biggest dent, in cells (1 to 8). Smaller dents, down to a third of this radius, are mixed in. A bigger Size gives fewer and wider
// rok_tip Size : hits, because the hit spots lie 2.5 times Size plus 2 cells apart. A hit finds the hull only within Size plus 2 cells of its spot (at most 10). Only
// rok_tip Size : filled cells change, so an empty model stays empty.
// rok_tip Amount : How many of the hit spots hold a hit, in percent. 0 gives no hit and no change. 100 puts a hit on every spot. The spots lie 2.5 times Size plus 2
// rok_tip Amount : cells apart. A hit with no hull within Size plus 2 cells does nothing, so the real number of hits is lower on a model with much open space.
// rok_tip Scorch : How far the soot spreads past the dent, from 0 to 100. The soot reaches 1 plus Scorch/40 dent radii from the dent centre. At 0 only the dent itself
// rok_tip Scorch : changes. At 100 the soot reaches 3.5 radii. Dark soot lies near the dent and ash flecks lie further out. Large values make the soot of near hits run
// rok_tip Scorch : together.
// rok_tip Glow : How many of the hits still burn, in percent. A burning hit paints about three quarters of the cells of its dent wall in the glow colour. 0 gives no
// rok_tip Glow : glow. 100 makes every dent glow. This changes colours only, not the shape.
// rok_tip Holes : 0 or 1. With 0 a dent stops one cell short of the far side, so a thin wall keeps one layer. With 1 a hit can cut right through. This matters only
// rok_tip Holes : where the hull is thinner than about 1.5 times Size. On a solid or thick model both values give the same result.
// rok_tip Seed : Picks another random layout of the hits, of the same kind. A model smaller than the hit spacing (2.5 times Size plus 2 cells) gets only a few hits, so
// rok_tip Seed : seeds then differ a lot.

// Battle damage: shots that hit the surface of a model. Each hit cuts a
// round dent into the hull and leaves soot round it.
// The hits sit on a grid in space. A grid cell is two and a half times
// Size wide, and it holds a hit as often as Amount says. A hit looks along
// the six axes for the nearest surface and moves onto it, so most hits
// land on the model. Size is the radius of the biggest dent. Smaller hits,
// like bullet holes, are mixed in.
// Scorch is how far the soot spreads past the dent. The dent is dark, a
// black ring goes round it, and dark and ash grey flecks thin out past
// that. Glow is how many hits still burn: the wall of the dent glows orange.
// With Holes 0 a dent never cuts through a wall. It stops one cell short of
// the far side, so a thin hull keeps its shape. With Holes 1 a shot can
// punch right through.
// Soot, dark, ash and glow are the colours of the rok_colours line.

float bd_hash(vec3 p3) {
  p3 = fract(p3 * 0.1031);
  p3 += dot(p3, p3.zyx + 31.32);
  return fract((p3.x + p3.y) * p3.z);
}

vec3 bd_dir(float n) {
  float axis = floor(n * 0.5);
  float sgn = mod(n, 2.0) < 0.5 ? 1.0 : -1.0;
  return axis < 0.5 ? vec3(sgn, 0.0, 0.0) : (axis < 1.5 ? vec3(0.0, sgn, 0.0) : vec3(0.0, 0.0, sgn));
}

float map(vec3 v) {
  float here = voxel(v);
  if (here == 0.0) return 0.0;  // the shader only cuts and paints
  vec3 c = floor(v) + 0.5;
  float R = m_size;
  float G = 2.5 * R + 2.0;  // the grid of hits: two cells reach past the farthest soot
  vec3 g0 = floor(c / G);
  float reach = R + 2.0;    // how far a hit looks for the hull
  float edge = 1.0 + m_scorch / 40.0;  // soot reaches this far, in dent radii
  float range = reach + R * (0.5 + edge) + 1.0;  // the farthest a hit can paint
  // The nearest hit's dent, soot and glow at this cell.
  float best = 1e9;         // distance to a dent, in units of its radius
  bool glow = false;
  float fleck = 0.0;
  bool cut = false;
  for (int i = 0; i < 125; i++) {
    // The grid cells two round this one: they hold every hit in range.
    vec3 g = g0 + vec3(mod(float(i), 5.0), mod(floor(float(i) / 5.0), 5.0), floor(float(i) / 25.0)) - 2.0;
    vec3 id = g + m_seed * 0.13;
    if (bd_hash(id) * 100.0 >= m_amount) continue;
    vec3 p = (g + vec3(bd_hash(id + 1.1), bd_hash(id + 2.2), bd_hash(id + 3.3))) * G;
    p = floor(p) + 0.5;
    if (length(c - p) > range) continue;
    // Move to the nearest surface along one axis: to the hull from open
    // space, or out of the hull from inside it.
    bool inside = voxel(p) != 0.0;
    float found = -1.0;
    float far = 0.0;
    for (int k = 1; k <= 10; k++) {
      if (float(k) > reach || found >= 0.0) break;
      for (int n = 0; n < 6; n++) {
        if (found < 0.0 && (voxel(p + bd_dir(float(n)) * float(k)) != 0.0) != inside) { found = float(n); far = float(k); }
      }
    }
    if (found < 0.0) continue;
    // e is the open cell on the surface, into the way the shot goes.
    vec3 into = inside ? -bd_dir(found) : bd_dir(found);
    vec3 e = p + bd_dir(found) * (inside ? far : far - 1.0);
    float r = max(1.0, floor(R * (0.35 + 0.65 * bd_hash(id + 4.4)) + 0.5));
    // The dent's centre: half a radius into the hull, so the dent is a bowl.
    vec3 centre = e + into * (0.5 * r);
    vec3 off = c - centre;
    float dist = length(off) / r;
    // Inside the dent the cell goes. With Holes 0 only when there is hull
    // behind it along the shot, so a wall keeps at least one cell.
    if (dist < 1.0 && !cut) cut = m_holes > 0.5 || voxel(c + into) != 0.0;
    if (dist < best) {
      best = dist;
      glow = bd_hash(id + 5.5) * 100.0 < m_glow;
      fleck = bd_hash(c + id * 3.7);
    }
  }
  if (best >= 1e8) return here;
  if (cut) return 0.0;
  if (best < 1.25 && glow && fleck < 0.75) return i_color_index + 3.0;  // the burning wall of the dent
  if (best < 1.2) return fleck < 0.75 ? i_color_index + 1.0 : i_color_index;  // the dent's wall
  float t = (best - 1.2) / max(edge - 1.2, 0.01);  // 0 at the dent, 1 where the soot ends
  if (t >= 1.0) return here;
  if (t < 0.3) return fleck < 0.85 ? i_color_index : i_color_index + 1.0;    // black soot
  if (t < 0.6) return fleck < 1.4 - t * 2.0 ? i_color_index + 1.0 : (fleck < 0.9 ? i_color_index + 2.0 : here);
  return fleck < 0.6 * (1.0 - t) / 0.4 ? i_color_index + 2.0 : here;           // ash flecks
}
