// rok_colours : { start = '0'  rgb = '1F4F8C 2E7FBF 5CC6D6 E8D9A0 6BAA3C 3F7A35 2C5A2A D9BA7A 8A8078 F2F4F7 A8B5A0' }
// rok_tip Height : How tall the highest ground is, in percent. The full height is the grid height, but never more than 30 percent of the smaller of grid width and depth.
// rok_tip Height : The sea level is about a quarter of the height, so Height also changes how deep the sea is. A taller grid adds nothing past that limit.
// rok_tip Land : How much of the map is land, from 0 to 100. At 0 there is mostly sea with a little land. At 100 there is mostly land.
// rok_tip Land : The shader keeps cells already there, so use it on an empty grid.
// rok_tip Mountains : How high the mountain ranges stand, from 0 to 100. At 0 there are no ranges, only plains.
// rok_tip Mountains : Ranges stand in belts well inland, so a low Land value gives few ranges.
// rok_tip Rivers : How many rivers there are, from 0 to 100. At 0 there are none. Rivers are thin bands and run only on the plains.
// rok_tip Rivers : On a 24 grid they are only a few cells (3 cells at 100), so check this on a grid of 64 or wider.
// rok_tip Climate : How warm the world is, from 0 to 100. Cold lands (low values, and the north, and high ground) get snow, tundra and dark forest.
// rok_tip Climate : Warm lands (high values) get more desert and grass, and less snow.
// rok_tip Forest : How much of the wet land has trees, from 0 to 100. At 0 almost none, at 100 most of the wet land.
// rok_tip Forest : Trees stand 1 or 2 cells above the ground. Desert, rock, snow and tundra never get trees.
// rok_tip Detail : How broken the coasts and ranges are, from 0 to 100. At 0 they are smooth, at 100 ragged.
// rok_tip Detail : The change is small: a few hundred cells on a 64 grid.
// rok_tip Seed : Picks another random layout of the same kind, from 0 to 9999.
// Rok puts these colours in the palette from the selected colour on: deep water,
// water, shallows, sand, grass, forest, dark forest, desert, rock, snow, tundra.

// A miniature map of a whole region, made to fit the grid: one or a few
// landmasses in a sea, with coasts, bays and islands, mountain ranges,
// rivers that cut the plains, and lakes. What grows follows the climate, as it
// does on a globe: snow and tundra in the cold north and on the heights, forest
// and dark pine forest in the wet country, grassland between, desert in the
// dry warm south, sand along the beaches, bare rock on the steep slopes. Every
// feature is sized by the grid, not in cells, so a bigger grid gives a bigger
// world and not more small bumps. Land is how much of the map is land,
// Mountains how high the ranges stand, Rivers how many there are, Climate
// how warm the world is, Forest how much of the wet land is wooded, Detail how
// broken the coasts and ranges are. The forest stands a little above the
// ground, so the woods read as woods. The voxels already there stay.

float wm_hash(vec2 p) {
  vec3 p3 = fract(vec3(p.x, p.y, p.x) * 0.1031);
  p3 += dot(p3, p3.yzx + 33.33);
  return fract((p3.x + p3.y) * p3.z);
}

float wm_noise(vec2 p) {
  vec2 i = floor(p);
  vec2 f = p - i;
  vec2 u = f * f * f * (f * (f * 6.0 - 15.0) + 10.0);
  float a = wm_hash(i);
  float b = wm_hash(i + vec2(1.0, 0.0));
  float c = wm_hash(i + vec2(0.0, 1.0));
  float d = wm_hash(i + vec2(1.0, 1.0));
  return mix(mix(a, b, u.x), mix(c, d, u.x), u.y);
}

float wm_fbm(vec2 p, float detail) {
  float sum = 0.0;
  float amp = 0.5;
  float norm = 0.0;
  float gain = mix(0.35, 0.55, detail);
  for (int k = 0; k < 5; k++) {
    sum += amp * wm_noise(p);
    norm += amp;
    amp *= gain;
    p = p * 2.03 + vec2(31.7, 11.3);
  }
  return sum / norm;
}

float wm_span() { return max(min(i_volume_size.x, i_volume_size.y), 1.0); }

// The tallest the ground gets, in cells.
float wm_top() { return min(i_volume_size.z, 0.3 * wm_span()) * m_height / 100.0; }

// The sea level, in cells.
float wm_sea() { return max(2.0, floor(wm_top() * 0.26)); }

// A column's place on the map: 0..1 across the smaller side, with the seed.
vec2 wm_p(vec2 column) { return (column + 0.5) / wm_span() + vec2(m_seed * 0.173, m_seed * 0.071); }

// The continent value of a column: high on land, low at sea.
float wm_land(vec2 column) {
  float det = m_detail / 100.0;
  vec2 p = wm_p(column);
  vec2 centre = (0.5 * i_volume_size.xy) / wm_span() + vec2(m_seed * 0.173, m_seed * 0.071);
  vec2 q = (p - centre) / ((0.5 * i_volume_size.xy) / wm_span());  // -1..1 over the map
  // The sea pulls in toward the map's edge, so the land lies in the middle.
  float edge = pow(clamp(length(q * vec2(0.95, 0.95)), 0.0, 1.0), 2.2);
  vec2 w = (vec2(wm_noise(p * 3.1 + vec2(3.7, 8.3)), wm_noise(p * 3.1 + vec2(11.3, 1.9))) - 0.5) * 0.35;
  float n = 0.62 * wm_fbm((p + w) * 2.3 + vec2(5.0, 9.0), det) + 0.38 * wm_fbm((p + w) * 5.0 + vec2(21.0, 3.0), det);
  return n * 1.3 - edge * 0.42 + 0.1;
}

float wm_thr() { return mix(0.68, 0.14, m_land / 100.0); }

// The height of the ground of a column, in cells, without rivers and woods.
float wm_ground(vec2 column) {
  float sea = wm_sea();
  float top = wm_top();
  float L = wm_land(column);
  float thr = wm_thr();
  if (L < thr) {
    // The sea floor: shallow on the shelf along the coast, deep further out.
    float depth = smoothstep(thr - 0.22, thr, L);
    return max(1.0, sea - 1.0 - (1.0 - depth) * (sea - 2.0));
  }
  float inland = clamp((L - thr) / 0.3, 0.0, 1.0);
  float det = m_detail / 100.0;
  vec2 p = wm_p(column);
  // Ranges stand where a slow noise is high, well inland; each is a chain of ridges.
  float belt = smoothstep(0.44, 0.64, wm_fbm(p * 2.4 + vec2(9.0, 4.0), det)) * smoothstep(0.08, 0.5, inland);
  vec2 wp = p + (vec2(wm_noise(p * 6.0 + vec2(1.0, 7.0)), wm_noise(p * 6.0 + vec2(8.0, 2.0))) - 0.5) * 0.12;
  float ridge = 1.0 - abs(2.0 * wm_noise(wp * 7.0 + vec2(17.0, 5.0)) - 1.0);
  ridge = pow(ridge, 1.7) * (0.65 + 0.35 * wm_noise(wp * 15.0));
  float plains = pow(inland, 0.6) * 0.24 + (wm_fbm(p * 6.0 + vec2(4.0, 12.0), det) - 0.5) * 0.06 * inland;
  float h = plains + belt * (0.12 + 0.88 * ridge) * (m_mount / 100.0) * 1.05;
  return sea + 1.0 + clamp(h, 0.0, 1.0) * (top - sea - 1.0);
}

// 1 where a river runs: thin winding bands, only on the plains, never on the heights.
float wm_river(vec2 column, float ground) {
  if (m_rivers < 0.5) return 0.0;
  float sea = wm_sea();
  if (ground <= sea + 1.0 || ground > sea + 1.0 + (wm_top() - sea) * 0.42) return 0.0;
  vec2 p = wm_p(column);
  vec2 w = (vec2(wm_noise(p * 4.0 + vec2(2.0, 6.0)), wm_noise(p * 4.0 + vec2(9.0, 3.0))) - 0.5) * 0.3;
  float d = abs(wm_fbm((p + w) * 3.6 + vec2(40.0, 13.0), 0.3) - 0.5);
  return d < m_rivers / 100.0 * 0.05 ? 1.0 : 0.0;
}

// The biome of a land column: 4 grass, 5 forest, 6 dark forest, 7 desert,
// 8 rock, 9 snow, 10 tundra, 3 sand. `ground` and `slope` are the column's.
float wm_biome(vec2 column, float ground, float slope) {
  float sea = wm_sea();
  float top = max(wm_top(), sea + 2.0);
  float alt = clamp((ground - sea - 1.0) / max(top - sea - 1.0, 1.0), 0.0, 1.0);
  vec2 p = wm_p(column);
  vec2 centre = (0.5 * i_volume_size.xy) / wm_span() + vec2(m_seed * 0.173, m_seed * 0.071);
  float north = clamp(((p - centre).y / ((0.5 * i_volume_size.y) / wm_span())) * 0.5 + 0.5, 0.0, 1.0);  // 0 south, 1 north
  float temp = 0.3 + 0.75 * m_climate / 100.0 - 0.62 * north - alt * 0.6;
  float moist = wm_fbm(p * 3.0 + vec2(31.0, 7.0), 0.4) + (wm_land(column) < wm_thr() + 0.12 ? 0.08 : 0.0);
  float speck = wm_noise(column / 2.0 + vec2(m_seed));
  if (wm_land(column) < wm_thr() + 0.035 + speck * 0.03) return 3.0;  // the beach, a thin band at the coast
  if (temp < 0.06 + (speck - 0.5) * 0.06) return 9.0;
  if (alt > 0.6 && slope >= 2.0) return 8.0;
  if (slope >= 3.0) return 8.0;
  if (temp < 0.26) return moist > 0.52 ? 6.0 : 10.0;
  if (moist < 0.47 && temp > 0.55) return 7.0;
  float wet = moist + (m_forest - 50.0) / 100.0 * 0.5;
  if (wet > 0.6) return temp < 0.42 ? 6.0 : 5.0;
  return moist < 0.46 && temp > 0.6 ? 7.0 : 4.0;
}

float map(vec3 v) {
  if (voxel(v) > 0.0) return voxel(v);
  vec2 cell = floor(v.xy);
  vec2 column = cell;
  float z = floor(v.z);
  float sea = wm_sea();
  float base = floor(wm_ground(column));
  float slope = max(
    max(abs(floor(wm_ground(column + vec2(1.0, 0.0))) - base), abs(floor(wm_ground(column - vec2(1.0, 0.0))) - base)),
    max(abs(floor(wm_ground(column + vec2(0.0, 1.0))) - base), abs(floor(wm_ground(column - vec2(0.0, 1.0))) - base)));
  if (base < sea) {
    // The sea: the floor, and water above it.
    if (z >= sea) return 0.0;
    if (z >= base) {
      float depth = sea - base;
      if (depth <= 2.0) return i_color_index + 2.0;           // the shelf, light
      return z >= sea - 2.0 ? i_color_index + 1.0 : i_color_index;
    }
    return z >= base - 2.0 ? i_color_index + 3.0 : i_color_index + 8.0;  // sand on the floor, rock below
  }
  float river = wm_river(column, base);
  float biome = wm_biome(column, base, slope);
  float ground = base;
  if (river > 0.5) ground = base - 1.0;
  // The woods stand above the ground by one or two cells.
  if (river < 0.5 && (biome == 5.0 || biome == 6.0)) ground += 1.0 + floor(wm_hash(column + vec2(m_seed)) * 2.0);
  if (z >= ground) return 0.0;
  if (river > 0.5 && z == ground - 1.0) return i_color_index + 1.0;
  if (z >= ground - 2.0) {
    if (river > 0.5) return i_color_index + 3.0;
    // A speckle of a neighbouring green keeps a meadow from being one flat colour.
    if (biome == 4.0 && wm_noise(column / 3.0 + vec2(m_seed, 4.0)) > 0.66) return i_color_index + 5.0 - (wm_hash(column) > 0.5 ? 1.0 : 0.0);
    return i_color_index + biome;
  }
  return i_color_index + 8.0;  // rock under everything
}
